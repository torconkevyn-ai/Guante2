    Código Arduino / ESP32 de referencia

    // 5 sensores flex en pines analógicos (ajusta a tu placa)
const int pines[5] = {A0, A1, A2, A3, A4};
void setup() { Serial.begin(115200); }
void loop() {
  for (int i = 0; i < 5; i++) {
    Serial.print(analogRead(pines[i]));
    if (i < 4) Serial.print(',');
  }
  Serial.println();
  delay(30);
}
